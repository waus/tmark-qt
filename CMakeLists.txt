cmake_minimum_required(VERSION 3.21)
project(TmarkQml VERSION 0.1.0 LANGUAGES CXX)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

add_library(tmark_codec src/codec.cpp)
add_library(Tmark::Codec ALIAS tmark_codec)
target_compile_features(tmark_codec PUBLIC cxx_std_20)
target_include_directories(tmark_codec PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(tmark_codec PROPERTIES EXPORT_NAME Codec)
if(EMSCRIPTEN)
    target_compile_options(tmark_codec PRIVATE -fexceptions)
endif()

set(TMARK_QT_COMPONENTS Core Gui Network Qml Quick QuickControls2 QuickEffects Lottie)
if(NOT EMSCRIPTEN)
    list(APPEND TMARK_QT_COMPONENTS Multimedia)
endif()
find_package(Qt6 6.5 QUIET COMPONENTS ${TMARK_QT_COMPONENTS})
if(Qt6_FOUND)
    if(EMSCRIPTEN)
        set(TMARK_MEDIA_QML qml/wasm/TmarkMedia.qml)
    else()
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(TMARK_FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libswscale)
        set(TMARK_MEDIA_QML qml/TmarkMedia.qml)
    endif()
    string(JOIN " " TMARK_QT_COMPONENTS_STRING ${TMARK_QT_COMPONENTS})
    set(TMARK_QT_DEPENDENCY "find_dependency(Qt6 6.5 COMPONENTS ${TMARK_QT_COMPONENTS_STRING})")
    qt_standard_project_setup(REQUIRES 6.5)
    if(Qt6_VERSION VERSION_GREATER_EQUAL 6.8)
        qt_policy(SET QTP0004 NEW)
    endif()
    set(TMARK_QML_FILES qml/TmarkView.qml qml/TmarkViewTheme.qml qml/TmarkRichText.qml qml/TmarkBlock.qml
        qml/TmarkCaption.qml ${TMARK_MEDIA_QML} qml/TmarkGallery.qml qml/TmarkTable.qml qml/TmarkList.qml)
    foreach(file IN LISTS TMARK_QML_FILES)
        get_filename_component(name "${file}" NAME)
        set_source_files_properties("${file}" PROPERTIES QT_RESOURCE_ALIAS "${name}")
    endforeach()
    if(EMSCRIPTEN)
        qt_add_library(tmark_qml STATIC)
    else()
        qt_add_library(tmark_qml SHARED)
    endif()
    qt_add_qml_module(tmark_qml
        URI io.github.waus.tmark
        VERSION 1.0
        OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/io/github/waus/tmark
        SOURCES src/viewmodel.hpp src/viewmodel.cpp src/mediasource.hpp src/mediasource.cpp
                src/medialoader.hpp src/medialoader.cpp src/mediaformat.hpp src/mediaformat.cpp
        QML_FILES ${TMARK_QML_FILES}
    )
    target_include_directories(tmark_qml PRIVATE src)
    if(EMSCRIPTEN)
        target_compile_options(tmark_qml PRIVATE -fexceptions)
    endif()
    target_link_libraries(tmark_qml PUBLIC tmark_codec Qt6::Core Qt6::Gui Qt6::Network Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::QuickEffects Qt6::Lottie)
    if(NOT EMSCRIPTEN)
        target_link_libraries(tmark_qml PUBLIC Qt6::Multimedia PRIVATE PkgConfig::TMARK_FFMPEG)
    endif()
    add_library(Tmark::Qml ALIAS tmark_qml)
    set_target_properties(tmark_qml PROPERTIES EXPORT_NAME Qml)

    option(TMARK_BUILD_EXAMPLE "Build the QML example" OFF)
    if(TMARK_BUILD_EXAMPLE)
        qt_add_executable(tmark_example example/main.cpp)
        set_source_files_properties(example/Main.qml PROPERTIES QT_RESOURCE_ALIAS Main.qml)
        qt_add_qml_module(tmark_example URI TmarkExample VERSION 1.0 QML_FILES example/Main.qml)
        target_link_libraries(tmark_example PRIVATE tmark_qml Qt6::Quick)
        if(EMSCRIPTEN)
            target_link_options(tmark_example PRIVATE -sNO_DISABLE_EXCEPTION_CATCHING)
        endif()
    endif()
else()
    message(STATUS "Qt 6.5+ unavailable: building the C++ codec only")
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(tmark_codec_test tests/codec_test.cpp)
    target_link_libraries(tmark_codec_test PRIVATE tmark_codec)
    target_compile_definitions(tmark_codec_test PRIVATE TMARK_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    add_test(NAME codec COMMAND tmark_codec_test)
    if(Qt6_FOUND)
        add_executable(tmark_media_test tests/media_test.cpp)
        target_include_directories(tmark_media_test PRIVATE src)
        target_link_libraries(tmark_media_test PRIVATE tmark_qml)
        add_test(NAME media COMMAND tmark_media_test)
        add_executable(tmark_qml_test tests/qml_test.cpp)
        target_link_libraries(tmark_qml_test PRIVATE tmark_qml Qt6::Qml Qt6::Quick)
        target_compile_definitions(tmark_qml_test PRIVATE TMARK_QML_BUILD_ROOT="${CMAKE_CURRENT_BINARY_DIR}")
        target_compile_definitions(tmark_qml_test PRIVATE TMARK_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
        add_test(NAME qml COMMAND tmark_qml_test)
        set_tests_properties(qml PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
endif()

install(TARGETS tmark_codec EXPORT TmarkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
if(Qt6_FOUND)
    if(EMSCRIPTEN)
        install(TARGETS tmark_qml_resources_1 tmark_qml_resources_2 tmark_qmlplugin_init
            EXPORT TmarkTargets OBJECTS DESTINATION ${CMAKE_INSTALL_LIBDIR}/tmark/objects)
    endif()
    install(TARGETS tmark_qml EXPORT TmarkTargets
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
        LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
    install(TARGETS tmark_qmlplugin EXPORT TmarkTargets
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}/qml/io/github/waus/tmark
        LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/qml/io/github/waus/tmark)
    install(FILES ${CMAKE_CURRENT_BINARY_DIR}/io/github/waus/tmark/qmldir
        ${CMAKE_CURRENT_BINARY_DIR}/io/github/waus/tmark/tmark_qml.qmltypes
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/qml/io/github/waus/tmark)
    list(REMOVE_ITEM TMARK_QML_FILES ${TMARK_MEDIA_QML})
    install(FILES ${TMARK_QML_FILES}
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/qml/io/github/waus/tmark)
    install(FILES ${TMARK_MEDIA_QML}
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/qml/io/github/waus/tmark RENAME TmarkMedia.qml)
    install(FILES src/medialoader.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/tmark)
endif()
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT TmarkTargets NAMESPACE Tmark:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Tmark)
configure_package_config_file(cmake/TmarkConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/TmarkConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Tmark)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/TmarkConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/TmarkConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/TmarkConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Tmark)
